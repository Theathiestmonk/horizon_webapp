// ── Amount in Words (full, supports lakh/crore for INR) ──────────────────
40
function AMOUNTWORDS(n, curr) {
41
 if (!n || n === 0) return '';
42
 const ones = ['','ONE','TWO','THREE','FOUR','FIVE','SIX','SEVEN','EIGHT','NINE',
43
 'TEN','ELEVEN','TWELVE','THIRTEEN','FOURTEEN','FIFTEEN','SIXTEEN',
44
 'SEVENTEEN','EIGHTEEN','NINETEEN'];
45
 const tens = ['','','TWENTY','THIRTY','FORTY','FIFTY','SIXTY','SEVENTY','EIGHTY','NINETY'];
46
 function words(x) {
47
 if (x < 20) return ones[x] || '';
48
 if (x < 100) return tens[Math.floor(x/10)]+(x%10?' '+ones[x%10]:'');
49
 if (x < 1000) return ones[Math.floor(x/100)]+' HUNDRED'+(x%100?' AND '+words(x%100):'');
50
 if (x < 100000) return words(Math.floor(x/1000))+' THOUSAND'+(x%1000?' '+words(x%1000):'');
51
 if (x < 10000000)return words(Math.floor(x/100000))+' LAKH'+(x%100000?' '+words(x%100000):'');
52
 return words(Math.floor(x/10000000))+' CRORE'+(x%10000000?' '+words(x%10000000):'');
53
 }
54
 const whole = Math.floor(Math.abs(n));
55
 const cents = Math.round((Math.abs(n)-whole)*100);
56
 const c = (curr||'USD').toUpperCase();
57
 let result = c+' '+words(whole)+' ONLY';
58
 if (cents > 0) result = c+' '+words(whole)+' AND CENTS '+words(cents)+' ONLY';
59
 return result;
60
}

function onEdit(e) {
64
 const sheet = e.source.getActiveSheet();
65
 const name = sheet.getName();
66
67
 // Stock col H (assigned_to) filled → set RESERVED; cleared → AVAILABLE
68
 if (name === 'Stock' && e.range.getColumn() === 8) {
69
 const row = e.range.getRow();
70
 if (row < 4) return;
71
 const val = e.range.getValue();
72
 sheet.getRange(row, 7).setValue(val ? 'RESERVED' : 'AVAILABLE');
73
 }
74
75
 // Stock col I (ship_date) filled → set SHIPPED
76
 if (name === 'Stock' && e.range.getColumn() === 9) {
77
 const row = e.range.getRow();
78
 if (row < 4 || !e.range.getValue()) return;
79
 sheet.getRange(row, 7).setValue('SHIPPED');
80
 }
81
82
 // CONTROL product selector → auto-fill unit price from Products col M
83
 if (name === 'CONTROL' && e.range.getA1Notation() === CFG.productCell) {
84
 const product = sheet.getRange(CFG.productCell).getValue();
85
 if (!product) return;
86
 const prodId = product.split(' — ')[0];
87
 const prodData = SpreadsheetApp.getActiveSpreadsheet()
88
 .getSheetByName('Products').getDataRange().getValues();
89
 for (let i = 3; i < prodData.length; i++) {
90
 if (prodData[i][0] === prodId) {
91
 sheet.getRange(CFG.unitPriceCell).setValue(prodData[i][12]);
92
 break;
93
 }
94
 }
95
 }

 const ctrl = ss.getSheetByName('CONTROL');

103
 const errors = [], warnings = [];
104
105

106

107
 const customer = ctrl.getRange(CFG.customerCell).getValue();
108

109
 const qty = Number(ctrl.getRange(CFG.qtyCell).getValue()) || 0;
110
 const price = Number(ctrl.getRange(CFG.unitPriceCell).getValue()) || 0;
111
 const container= ctrl.getRange(CFG.containerCell).getValue();
112
 const mode = ctrl.getRange(CFG.modeCell).getValue();
113
114
 if (!inv) errors.push('Invoice number is required');
115
 if (!date) errors.push('Invoice date is required');
116
 if (!customer) errors.push('Customer is required');
117
 if (!product) errors.push('Product is required');
118
 if (qty <= 0) errors.push('Quantity must be > 0');
119
 if (price <= 0)errors.push('Unit price must be > 0');
120
 if (!container)errors.push('Container number is required');
121
 if (!/^[A-Z]{4}\d{7}$/.test(String(container).toUpperCase()))
122
 errors.push('Container format invalid (4 letters + 7 digits)');
123
124
 if (mode === 'FINAL') {
125
 const vins = stock.getRange('A4:H2000').getValues()
126
 .filter(r => r[7] === inv);
127
 if (vins.length === 0)
128
 errors.push('No vehicles assigned to this invoice (assign via sidebar)');
129
 else if (vins.length !== qty)
130
 errors.push(`VIN count ${vins.length} ≠ quantity ${qty}`);
131
 const missing = vins.filter(r => !r[0]).length;
132
 if (missing > 0)
133
 errors.push(`${missing} assigned vehicle(s) have empty chassis numbers`);
134
 }
135
136
 return { errors, warnings };
137
}

 stock.getRange(i+4,7).setValue('RESERVED');

 stock.getRange(i+4,8).setValue(invoiceNo);

 assigned++;

 stock.getRange(i+4,7).setValue('AVAILABLE');
296
 stock.getRange(i+4,8).setValue('');

 unassigned++;
298
 }
299
 });
300
 logAudit(invoiceNo,'VEHICLES_ASSIGNED',`${assigned} assigned, ${unassigned} unassigned`);
